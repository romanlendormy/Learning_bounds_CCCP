\newcommand{\plotN}[6]{%
  \addplot+[#5, color=#4, mark options={fill=#4}, unbounded coords=discard,
    x filter/.expression={\thisrow{N}==#3 ? x : nan}] table[col sep=comma, x=n, y=#2] {#1};
  \if\relax\detokenize{#6}\relax\else
  \addplot+[lb, color=#4, mark options={fill=white},
    x filter/.expression={(\thisrow{N}==#3 && \thisrow{#6}==1) ? x : nan}] table[col sep=comma, x=n, y=#2] {#1};
  \fi}
